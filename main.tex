def normalize_scores(scores):
    """Scale a list of scores to the range [0, 1]."""
    # 中文注释
    lo = min(scores)
    hi = max(scores)
    if hi == lo:
        return [0.0 for _ in scores]

    return [(score - lo) / (hi - lo) for score in scores]

values = [72, 88, 91, 67]
print(normalize_scores(values))
